cmake_minimum_required(VERSION 3.20)

project(DSPSim LANGUAGES CXX C)

find_package(glfw3 REQUIRED)

set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

add_executable(DSPSim src/main.cpp src/engine/engine.cpp src/window/window.cpp src/render/render.cpp src/glad.c)

target_link_libraries(DSPSim PRIVATE glfw)

target_include_directories(
    DSPSim
    PRIVATE
    ${PROJECT_SOURCE_DIR}/include
)

target_compile_features(DSPSim PRIVATE cxx_std_20)

target_compile_options(
    DSPSim
    PRIVATE
        -Wall
        -Werror
        -Wextra
)
