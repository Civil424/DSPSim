cmake_minimum_required(VERSION 3.20)

project(DSPSim LANGUAGES CXX)

add_executable(DSPSim src/main.cpp)

target_include_directories(
    DSPSim
    PRIVATE
    ${PROJECT_SOURCE_DIR}/include
)

target_compile_features(DSPSim PRIVATE cxx_std_20)

target_compile_options(
    DSPSim
    PRIVATE
        -Wall
        -Werror
        -Wextra
)
